\sausbox{name=intro,column=1,below=top}{Why one identity}{%
  Slides, posters and documents from every department should be recognisably
  \sauskey{The Shaikh Ayaz University}. A shared brand layer,
  \texttt{saus-brand.sty}, now drives both the Beamer theme and this poster
  class, so the two cannot drift apart.
  \begin{itemize}
    \item Colours sampled from the university crest.
    \item One type family with matching mathematics.
    \item English, Sindhi and Urdu set side by side.
  \end{itemize}}

\sauscard{name=access,column=1,below=intro}{Contrast and accessibility}{%
  Every text colour is checked against WCAG 2.x. For relative luminances
  $L_1 > L_2$ the contrast ratio is
  \[
    C = \frac{L_1 + 0.05}{L_2 + 0.05}.
  \]
  \begin{tabular}{@{}llr@{}}
    \toprule
    \textbf{Text} & \textbf{Background} & \textbf{Ratio} \\
    \midrule
    Ink        & White       & 16.7\,:\,1 \\
    Maroon     & White       & 10.2\,:\,1 \\
    Royal blue & White       & 13.1\,:\,1 \\
    Gold       & Deep maroon &  5.4\,:\,1 \\
    Gold       & White       &  2.6\,:\,1 \\
    \bottomrule
  \end{tabular}\par
  AA needs 4.5\,:\,1 for body text, so gold is used on deep maroon,
  never on white.}

\sausbox{name=rules,column=3,between=slides and bottom}{Brand rules in brief}{%
  \begin{itemize}
    \item Use the crest unaltered: colour on light grounds, white line-art on
          maroon or photographs.
    \item Maroon leads; blue supports; gold accents and never carries body text
          on white.
    \item One type family. Headings never hyphenate; text is ragged-right.
    \item Keep the motto in its own voice:
          {\sausmottofont Revival of Educational Glory of Shikarpur.}
  \end{itemize}}
